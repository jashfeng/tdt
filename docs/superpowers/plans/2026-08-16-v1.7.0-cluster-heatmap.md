# v1.7.0 标记聚合 + 热力图 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 beige-tdt-map 插件（uni-app x 蒸汽模式天地图原生插件）新增标记聚合（MarkerCluster）与热力图（Heatmap）两大数据可视化能力，四端统一 API。

**Architecture:** 四端全部使用各平台原生聚合/热力能力（Android/iOS/Web 走 MapLibre GeoJSONSource cluster + 样式图层；鸿蒙走 MapKit 原生 `addClusterOverlay`/`addHeatmap`），interface.uts 新增 4 个 API，uvue 组件层加 thin wrapper，点击聚合簇自动放大展开。所有修改四副本同步（主副本 / example-project / tmp-publish / C 盘 tdt-app，C 盘 config.uts 跳过）。

**Tech Stack:** MapLibre Android 11.8.1（GeoJsonSource cluster + CircleLayer/SymbolLayer + HeatmapLayer）、MapLibre iOS（MLNShapeSource clustering + MLNCircleStyleLayer + MLNHeatmapStyleLayer）、华为 MapKit（ClusterOverlay + Heatmap）、MapLibre GL JS 4.7.1（cluster source + circle/symbol/heatmap layer）、UTS / uvue。

```uts
// ═══════════════════════════════════════════════════════════
// v1.7.0: 标记聚合（context7: MapLibre 官方 create-and-style-clusters 示例）
// cluster:true 后引擎自动生成 point_count / cluster_id 属性；
// 三个图层：聚合簇圆点 / 聚合簇数量文字 / 未聚合原始点
// ═══════════════════════════════════════════════════════════
let clusterLayerIds: Array<string> = []

function removeClusterInternal(): void {
    if (mapInstance == null) return
    const mlg = getMapLibre()
    if (mlg == null) return
    // 移除 click 监听（幂等：off 未注册的监听不报错）
    mapInstance.off('click', 'tdt-cluster-layer', onClusterClick)
    for (let i = 0; i < clusterLayerIds.length; i++) {
        try { mapInstance.removeLayer(clusterLayerIds[i]) } catch (e) { /* ignore */ }
    }
    clusterLayerIds = []
    try { mapInstance.removeSource('tdt-cluster-source') } catch (e) { /* ignore */ }
}

function onClusterClick(e: any): void {
    try {
        const features = mapInstance?.queryRenderedFeatures(e.point, { layers: ['tdt-cluster-layer'] })
        if (features == null || features.length == 0) return
        const clusterId = features[0].properties.cluster_id
        if (clusterId == null) return
        const src = mapInstance?.getSource('tdt-cluster-source') as any
        if (src == null) return
        src.getClusterExpansionZoom(clusterId).then((zoom: number) => {
            const coords = features[0].geometry.coordinates
            mapInstance?.easeTo({ center: [coords[0], coords[1]], zoom: zoom })
        }).catch((err: any) => {
            console.error('TDTMap Web: getClusterExpansionZoom 失败', err)
        })
    } catch (err: any) {
        console.error('TDTMap Web: 聚合点击处理失败', err)
    }
}

export function addMarkerCluster(points: Array<MapClickDetail>, options?: ClusterOptions): void {
    if (mapInstance == null) return
    const mlg = getMapLibre()
    if (mlg == null) return
    removeClusterInternal()
    if (points.length == 0) return
    try {
        const features: Array<any> = []
        for (let i = 0; i < points.length; i++) {
            features.push({
                type: 'Feature',
                geometry: { type: 'Point', coordinates: [points[i].lng, points[i].lat] },
                properties: {}
            })
        }
        mapInstance.addSource('tdt-cluster-source', {
            type: 'geojson',
            data: { type: 'FeatureCollection', features: features },
            cluster: true,
            clusterRadius: options?.clusterRadius ?? 50,
            clusterMaxZoom: options?.clusterMaxZoom ?? 14
        })
        // 聚合簇圆点（step 表达式按数量分级着色/分级半径）
        mapInstance.addLayer({
            id: 'tdt-cluster-layer',
            type: 'circle',
            source: 'tdt-cluster-source',
            filter: ['has', 'point_count'],
            paint: {
                'circle-color': ['step', ['get', 'point_count'], '#51BBD6', 100, '#F1F075', 750, '#F28CB1'],
                'circle-radius': ['step', ['get', 'point_count'], 20, 100, 30, 750, 40],
                'circle-stroke-width': 1,
                'circle-stroke-color': '#FFFFFF'
            }
        })
        clusterLayerIds.push('tdt-cluster-layer')
        // 聚合簇数量文字
        mapInstance.addLayer({
            id: 'tdt-cluster-count',
            type: 'symbol',
            source: 'tdt-cluster-source',
            filter: ['has', 'point_count'],
            layout: {
                'text-field': '{point_count_abbreviated}',
                'text-size': 12
            }
        })
        clusterLayerIds.push('tdt-cluster-count')
        // 未聚合的原始点
        mapInstance.addLayer({
            id: 'tdt-cluster-unclustered',
            type: 'circle',
            source: 'tdt-cluster-source',
            filter: ['!', ['has', 'point_count']],
            paint: {
                'circle-color': '#11B4DA',
                'circle-radius': 5,
                'circle-stroke-width': 1,
                'circle-stroke-color': '#FFFFFF'
            }
        })
        clusterLayerIds.push('tdt-cluster-unclustered')
        mapInstance.on('click', 'tdt-cluster-layer', onClusterClick)
    } catch (e: any) {
        console.error('TDTMap Web: addMarkerCluster 失败', e)
    }
}

export function removeMarkerCluster(): void {
    removeClusterInternal()
}
```

```uts
// v1.7.0: 热力图（context7: MapLibre 官方 heatmap 图层示例）
// heatmap-weight 取数据 weight 属性，radius/opacity 可配
// ═══════════════════════════════════════════════════════════
export function addHeatmap(points: Array<HeatmapItem>, options?: HeatmapOptions): void {
    if (mapInstance == null) return
    removeHeatmap()
    if (points.length == 0) return
    try {
        const features: Array<any> = []
        for (let i = 0; i < points.length; i++) {
            features.push({
                type: 'Feature',
                geometry: { type: 'Point', coordinates: [points[i].lng, points[i].lat] },
                properties: { weight: points[i].weight ?? 1 }
            })
        }
        mapInstance.addSource('tdt-heatmap-source', {
            type: 'geojson',
            data: { type: 'FeatureCollection', features: features }
        })
        mapInstance.addLayer({
            id: 'tdt-heatmap-layer',
            type: 'heatmap',
            source: 'tdt-heatmap-source',
            paint: {
                'heatmap-weight': ['get', 'weight'],
                'heatmap-radius': options?.radius ?? 30,
                'heatmap-opacity': options?.opacity ?? 0.6,
                'heatmap-color': [
                    'interpolate', ['linear'], ['heatmap-density'],
                    0, 'rgba(0, 0, 255, 0)',
                    0.2, 'rgb(0, 0, 255)',
                    0.4, 'rgb(0, 255, 0)',
                    0.6, 'rgb(255, 255, 0)',
                    0.8, 'rgb(255, 165, 0)',
                    1, 'rgb(255, 0, 0)'
                ]
            }
        })
    } catch (e: any) {
        console.error('TDTMap Web: addHeatmap 失败', e)
    }
}

export function removeHeatmap(): void {
    if (mapInstance == null) return
    try { mapInstance.removeLayer('tdt-heatmap-layer') } catch (e) { /* ignore */ }
    try { mapInstance.removeSource('tdt-heatmap-source') } catch (e) { /* ignore */ }
}
```

```uts
// ══════════════════════════════════════════════
// v1.7.0 新增：标记聚合 + 热力图（数据可视化）
// context7: maplibre-native issue #3519 真实代码（Android 11.9.0）验证 GeoJsonSource cluster
// API 家族；已知竞态 bug（setGeoJson 更新后立即 getClusterExpansionZoom 崩溃）→
// 本实现重复调用 addMarkerCluster 时先 remove 再重建，规避该 bug
// ══════════════════════════════════════════════
let clusterSourceRef: GeoJsonSource | null = null
let clusterLayerRefs: Array<CircleLayer | SymbolLayer> = []
let heatmapSourceRef: GeoJsonSource | null = null
let heatmapLayerRef: HeatmapLayer | null = null

function buildPointFeatures(points: Array<MapClickDetail>): FeatureCollection {
    const features: Array<Feature> = []
    for (let i = 0; i < points.length; i++) {
        features.push(Feature.fromGeometry(Point.fromLngLat(points[i].lng.toDouble(), points[i].lat.toDouble())))
    }
    return FeatureCollection.fromFeatures(features)
}

// removeSource 统一传字符串 ID（Style.removeSource(String) 与 removeSource(Source) 重载均存在，
// 传 ID 最简且规避 UTS 重载解析歧义）
function removeClusterInternal(): void {
    const style = mapLibreMap?.getStyle()
    if (style != null) {
        for (let i = 0; i < clusterLayerRefs.length; i++) {
            try { style.removeLayer(clusterLayerRefs[i]) } catch (e: any) { /* ignore */ }
        }
        try { style.removeSource("tdt-cluster-source") } catch (e: any) { /* ignore */ }
    }
    clusterLayerRefs = []
    clusterSourceRef = null
}

export function addMarkerCluster(points: Array<MapClickDetail>, options?: ClusterOptions): void {
    const style = mapLibreMap?.getStyle()
    const errorCb = onErrorCallback
    if (style == null) {
        if (errorCb != null) errorCb({ code: 4, message: "样式未加载，无法添加聚合图层" })
        return
    }
    removeClusterInternal()
    if (points.length == 0) return
    try {
        const radius = options?.clusterRadius ?? 50
        const maxZoom = options?.clusterMaxZoom ?? 14
        const geoOpts = new GeoJsonOptions()
            .withCluster(true)
            .withClusterRadius(radius.toInt())
            .withClusterMaxZoom(maxZoom.toInt())
        const source = new GeoJsonSource("tdt-cluster-source", buildPointFeatures(points), geoOpts)
        style.addSource(source)
        clusterSourceRef = source

        // 聚合簇圆点（filter has("point_count") 区分聚合簇与原始点）
        const clusterLayer = new CircleLayer("tdt-cluster-layer", "tdt-cluster-source")
        clusterLayer.setProperties(
            PropertyFactory.circleColor(Color.parseColor("#51BBD6")),
            PropertyFactory.circleRadius(20.0),
            PropertyFactory.circleStrokeColor(Color.parseColor("#FFFFFF")),
            PropertyFactory.circleStrokeWidth(1.0)
        )
        clusterLayer.setFilter(Expression.has("point_count"))
        style.addLayer(clusterLayer)
        clusterLayerRefs.push(clusterLayer)

        // 聚合簇数量文字（point_count_abbreviated 为引擎自动生成的缩写计数）
        const countLayer = new SymbolLayer("tdt-cluster-count", "tdt-cluster-source")
        countLayer.setProperties(
            PropertyFactory.textField(Expression.get("point_count_abbreviated")),
            PropertyFactory.textSize(12.0),
            PropertyFactory.textColor(Color.parseColor("#FFFFFF"))
        )
        countLayer.setFilter(Expression.has("point_count"))
        style.addLayer(countLayer)
        clusterLayerRefs.push(countLayer)

        // 未聚合原始点（filter 非 has("point_count")）
        const unclusterLayer = new CircleLayer("tdt-cluster-unclustered", "tdt-cluster-source")
        unclusterLayer.setProperties(
            PropertyFactory.circleColor(Color.parseColor("#11B4DA")),
            PropertyFactory.circleRadius(5.0),
            PropertyFactory.circleStrokeColor(Color.parseColor("#FFFFFF")),
            PropertyFactory.circleStrokeWidth(1.0)
        )
        unclusterLayer.setFilter(Expression.not(Expression.has("point_count")))
        style.addLayer(unclusterLayer)
        clusterLayerRefs.push(unclusterLayer)
    } catch (e: any) {
        console.error("TDTMap: 添加聚合图层失败: " + e.message)
        if (errorCb != null) errorCb({ code: 4, message: "添加聚合图层失败: " + e.message })
        removeClusterInternal()
    }
}

export function removeMarkerCluster(): void {
    removeClusterInternal()
}
```

```uts
            props.addProperty("weight", (points[i].weight ?? 1).toDouble())
            features.push(Feature.fromGeometry(
                Point.fromLngLat(points[i].lng.toDouble(), points[i].lat.toDouble()),
                props
            ))
        }
        const source = new GeoJsonSource("tdt-heatmap-source", FeatureCollection.fromFeatures(features))
        style.addSource(source)
        heatmapSourceRef = source

        const layer = new HeatmapLayer("tdt-heatmap-layer", "tdt-heatmap-source")
        layer.setProperties(
            PropertyFactory.heatmapWeight(Expression.get("weight")),
            PropertyFactory.heatmapRadius((options?.radius ?? 30).toDouble()),
            PropertyFactory.heatmapOpacity((options?.opacity ?? 0.6).toDouble())
        )
        style.addLayer(layer)
        heatmapLayerRef = layer
    } catch (e: any) {
        console.error("TDTMap: 添加热力图层失败: " + e.message)
        if (errorCb != null) errorCb({ code: 4, message: "添加热力图层失败: " + e.message })
        removeHeatmap()
    }
}

    if (style != null) {
        if (heatmapLayerRef != null) {
            try { style.removeLayer(heatmapLayerRef) } catch (e: any) { /* ignore */ }
        }
        try { style.removeSource("tdt-heatmap-source") } catch (e: any) { /* ignore */ }
```

```javascript
    }));
    try { map.removeLayer('tdt-heatmap-layer'); } catch (e) {}
    try { map.removeSource('tdt-heatmap-source'); } catch (e) {}
    map.addSource('tdt-heatmap-source', { type: 'geojson', data: { type: 'FeatureCollection', features: features } });
    map.addLayer({ id: 'tdt-heatmap-layer', type: 'heatmap', source: 'tdt-heatmap-source',
        paint: {
            'heatmap-weight': ['get', 'weight'],
            'heatmap-radius': (options && options.radius) || 30,
            'heatmap-opacity': (options && options.opacity) || 0.6
        } });
    window.__tdtHeatmapAdded = true;
}
```

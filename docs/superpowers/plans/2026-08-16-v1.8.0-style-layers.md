## 实现语义（对 spec 的精确化，四端统一）

1. `setStyle(json)`：完全替换为自定义样式（天底图由用户样式自带，插件不自动加回）；样式加载完成后自动重建用户数据图层（自定义瓦片/WMS/聚合/热力）。JSON 前置校验：必须含 `"version"` 字段，否则报 code=4 不切换（防解析失败地图空白）。鸿蒙 mapError code=5。
2. `removeStyle()`：恢复插件默认天地图样式 + 重建用户数据图层；从未 setStyle 时幂等 no-op。
3. iOS 特例：天地图底图是"空样式 + didFinishLoadingStyle 动态加图层"架构，现有 delegate 会无条件加回天底图 → 必须用 `styleRebuildMode` 三态拦截，否则 setStyle 后天地图瓦片会盖住用户样式（真 bug 隐患，spec 自审发现）。
4. `zIndex` 语义收窄：仅支持 `0`（插到底图注记层之下）与 `undefined`（默认追加最上层）；其他值 console.warn 忽略。鸿蒙 zIndex 忽略+warn。
5. `options` 缺省值与现状完全一致（opacity=1、minZoom=1、maxZoom=18），不传 options 行为零变化。
6. `setMaxBounds` 任一参数 null 即清除：Android/Web 官方 API 传 null；iOS 一次性限制无持久状态 → null 分支 no-op；鸿蒙官方未确认 null 行为 → 全球范围（±180/±85.05）等效清除，不依赖未验证 API。
7. 鸿蒙 `setStyle`/`addWMSTileLayer` 显式 mapError `code=5`（平台不支持），不静默失败。
8. `addWMSTileLayer` 三端实现 = 拼装 GetMap 模板 URL（`{bbox-epsg-3857}` 为 style spec 标准占位符）+ 复用 addCustomTileLayer；状态无需区分 WMS（模板 URL 重建时按普通瓦片重放即正确）。

export function addCustomTileLayer(id: string, tileUrl: string, options?: TileLayerOptions): void {
    if (mapInstance == null) return
    const sourceId = 'custom-tile-src-' + id
    const layerId = 'custom-tile-layer-' + id
    try {
        mapInstance.addSource(sourceId, {
            type: 'raster',
            tiles: [tileUrl],
            tileSize: 256,
            minzoom: options?.minZoom ?? 1,
            maxzoom: options?.maxZoom ?? 18
        })
        const layerObj: any = {
            id: layerId,
            type: 'raster',
            source: sourceId,
            minzoom: options?.minZoom ?? 1,
            maxzoom: options?.maxZoom ?? 18,
            paint: { 'raster-opacity': options?.opacity ?? 1 }
        }
        if (options != null && options.zIndex == 0) {
            // zIndex=0: 插到天地图注记层之下（底图之上）
            mapInstance.addLayer(layerObj, 'tdt-anno-layer')
        } else {
            mapInstance.addLayer(layerObj)
        }
        if (options != null && options.zIndex != null && options.zIndex != 0) {
            console.warn('TDTMap Web: zIndex 仅支持 0（贴底）与 undefined（置顶），当前值已忽略')
        }
        customTileLayers.set(id, { sourceId: sourceId, layerId: layerId, tileUrl: tileUrl, options: options ?? {} })
        customTileLayerIds.push(id)
    } catch (e: any) {
        console.error('TDTMap Web: addCustomTileLayer 失败', e)
        if (mapErrorCallback != null) {
            mapErrorCallback({ code: 4, message: '添加瓦片图层失败: ' + (e.message ?? '') })
        }
    }
}

// context7 验证：{bbox-epsg-3857} 为 MapLibre style spec 标准占位符（style spec 官方 + 天地图官方文档确认）
function buildWmsTileUrl(wmsUrl: string, layers: string): string {
    const sep = wmsUrl.indexOf('?') >= 0 ? '&' : '?'
    return wmsUrl + sep + 'SERVICE=WMS&REQUEST=GetMap&VERSION=1.1.1&LAYERS=' + layers +
        '&STYLES=&FORMAT=image/png&TRANSPARENT=true&SRS=EPSG:3857&BBOX={bbox-epsg-3857}&WIDTH=256&HEIGHT=256'
}

export function addWMSTileLayer(id: string, wmsUrl: string, layers: string, options?: TileLayerOptions): void {
    addCustomTileLayer(id, buildWmsTileUrl(wmsUrl, layers), options)
}

function rebuildUserLayersAfterStyle(): void {
    const ids = customTileLayerIds.slice()
    customTileLayerIds = []
    const saved: Array<any> = []
    for (let i = 0; i < ids.length; i++) {
        const info = customTileLayers.get(ids[i])
        if (info != null) saved.push({ id: ids[i], info: info })
    }
    customTileLayers.clear()
    for (let i = 0; i < saved.length; i++) {
        addCustomTileLayer(saved[i].id, saved[i].info.tileUrl, saved[i].info.options)
    }
    if (clusterDataRef != null) {
        const pts = clusterDataRef
        const opts = clusterOptionsRef
        clusterDataRef = null
        clusterOptionsRef = null
        addMarkerCluster(pts, opts)
    }
    if (heatmapDataRef != null) {
        const pts = heatmapDataRef
        const opts = heatmapOptionsRef
        heatmapDataRef = null
        heatmapOptionsRef = null
        addHeatmap(pts, opts)
    }
}

export function setStyle(styleJson: string): void {
    if (mapInstance == null) return
    let parsed: any = null
    try {
        parsed = JSON.parse(styleJson)
    } catch (e: any) {
        console.error('TDTMap Web: setStyle JSON 解析失败', e)
        if (mapErrorCallback != null) {
            mapErrorCallback({ code: 4, message: 'setStyle JSON 解析失败' })
        }
        return
    }
    if (parsed == null || parsed.version == null) {
        if (mapErrorCallback != null) {
            mapErrorCallback({ code: 4, message: 'setStyle 需为含 version 字段的 MapLibre Style Spec JSON' })
        }
        return
    }
    mapInstance.setStyle(parsed)
    mapInstance.once('style.load', () => {
        rebuildUserLayersAfterStyle()
    })
}

export function removeStyle(): void {
    if (mapInstance == null) return
    const cfg = LAYER_CONFIG.get(currentMapType)
    if (cfg == null) return
    mapInstance.setStyle({ version: 8, glyphs: 'https://demotiles.maplibre.org/font/{fontstack}/{range}.pbf', sources: {}, layers: [] })
    mapInstance.once('style.load', () => {
        addTdtBaseLayers(mapInstance, cfg)
        rebuildUserLayersAfterStyle()
    })
}
